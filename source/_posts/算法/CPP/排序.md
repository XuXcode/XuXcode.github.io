# 排序

### 1.插入排序

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N=1e5+10;
int n;
int a[N];
void insert_sort(int n){
	//依次枚举待排序的元素 
	for(int i=2;i<=n;i++){ //第一个位置默认就是有序的 
		int key= a[i]; 
		 //前面比key大的，统一右移
		 int  j=i-1;
		 while(j>=1&&a[j]>key){
		 	a[j+1]=a[j];
		 	j--;
		 }
		 a[j+1]=key;
	}
}
int main(){
	cin>>n;
	for(int i=1;i<=n;i++) cin>>a[i];
	
	insert_sort();
	
	for(int i=i;i<=n;i++) cout<<a[i]<<" ";
	return 0;
} 
```
